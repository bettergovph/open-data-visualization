⚙️ Technical Implementation

Database Queries Required

-- Political HHI per province
SELECT province, 
       SUM(POWER(surname_count::float / total_positions, 2)) * 100 as political_hhi
FROM dynasty.political_dynasties 
GROUP BY province;

-- Contractor concentration per province  
SELECT area_of_delivery,
       SUM(POWER(contractor_share, 2)) as contractor_hhi
FROM philgeps.contracts 
GROUP BY area_of_delivery;
